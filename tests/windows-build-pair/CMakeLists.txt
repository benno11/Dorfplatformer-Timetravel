cmake_minimum_required(VERSION 3.16)
project(WindowsBuildPairRegression LANGUAGES C CXX)
if(NOT MSVC)
  message(FATAL_ERROR "This fixture tests MSVC PE section layout")
endif()
set(CMAKE_CXX_STANDARD 17)
get_filename_component(GAME_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
include("${GAME_ROOT}/cmake/BuildPairLink.cmake")
add_library(crypto STATIC
  "${GAME_ROOT}/third_party/monocypher/monocypher.c"
  "${GAME_ROOT}/third_party/monocypher/monocypher-ed25519.c")
target_include_directories(crypto PUBLIC "${GAME_ROOT}/third_party/monocypher")
add_executable(signer "${GAME_ROOT}/tools/BuildPairSigner.cpp" "${GAME_ROOT}/src/BuildPairIntegrity.cpp")
target_include_directories(signer PRIVATE "${GAME_ROOT}/src")
target_compile_definitions(signer PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(signer PRIVATE crypto bcrypt)
add_executable(pair_exe main.cpp "${GAME_ROOT}/src/BuildPairSection.cpp")
add_library(pair_dll SHARED "${GAME_ROOT}/src/BuildPairSection.cpp")
target_compile_definitions(pair_dll PRIVATE PLATFORMER_GAME_LIBRARY=1)
platformer_finalize_signed_pe(pair_exe)
platformer_finalize_signed_pe(pair_dll)
add_executable(integrity_test "${GAME_ROOT}/tests/BuildPairIntegrityTest.cpp" "${GAME_ROOT}/src/BuildPairIntegrity.cpp")
target_include_directories(integrity_test PRIVATE "${GAME_ROOT}/src")
# Assertions must execute in RelWithDebInfo too.
target_compile_options(integrity_test PRIVATE /UNDEBUG)
target_link_libraries(integrity_test PRIVATE crypto)
enable_testing()
add_test(NAME integrity COMMAND integrity_test)
add_test(NAME keygen COMMAND signer keygen "${CMAKE_CURRENT_BINARY_DIR}/signing.key" "${CMAKE_CURRENT_BINARY_DIR}/BuildPairTrust.h")
set_tests_properties(keygen PROPERTIES FIXTURES_SETUP key)
add_test(NAME seal COMMAND signer seal "${CMAKE_CURRENT_BINARY_DIR}/signing.key" $<TARGET_FILE:pair_exe> $<TARGET_FILE:pair_dll> full)
set_tests_properties(seal PROPERTIES FIXTURES_REQUIRED key FIXTURES_SETUP sealed)
add_test(NAME reseal COMMAND signer seal "${CMAKE_CURRENT_BINARY_DIR}/signing.key" $<TARGET_FILE:pair_exe> $<TARGET_FILE:pair_dll> full)
set_tests_properties(reseal PROPERTIES FIXTURES_REQUIRED sealed)
