cmake_minimum_required(VERSION 3.16)
project(AutoHop C)
add_library(auto_hop MODULE auto_hop.c)
target_include_directories(auto_hop PRIVATE ../include)
set_target_properties(auto_hop PROPERTIES PREFIX "")
if(APPLE)
  set_target_properties(auto_hop PROPERTIES SUFFIX ".dylib")
endif()
if(WIN32)
  target_sources(auto_hop PRIVATE exports.def)
endif()
